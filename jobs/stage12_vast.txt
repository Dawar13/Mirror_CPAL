# Rented RTX 5090 (vast.ai, 2026-11-09), after the A100 went offline. Launch from /workspace/Mirror_CPAL with
#   export HF_HOME=/workspace/hf MIRROR_MEM_SCALE=2.5; PAR=3 NEED_MB=6000 bash scripts/vm_queue.sh jobs/stage12_vast.txt
# The laptop uplink is slow, so the A100 artefacts (M1 structure, X2 operators for fit seeds 0 to 2, M3-pilot healed
# checkpoints) arrive one by one over several hours; each job waits for the files it needs (W = wait for files).
# ImageNet caches: cache/train_meta.json is written last. ADE20K "original" was started by hand before this queue.
# One healing job at a time (V-exact k = 6, then k = 12 again: the A100 checkpoint was not copied before that VM went
# offline; the rerun goes to runs/m3_vexact_5090/ so the committed A100 result stays untouched, and it doubles as a
# second-device check). M2 latency and profile run last, alone, by hand.
#
# ADE20K, MLP and V-exact variants (need the M1 structure and the fit seed 0 operators).
bash -c 'W="runs/m1/m1_structure.npz runs/x2/perlayer_reg_s0/results/e1_operators.npz"; for f in $W; do while [ ! -f $f ]; do sleep 60; done; done; python scripts/tier2_ade20k.py --model dinov2_vitb14_reg --variants vexact_k6,mirror_mlp_k6,svd_mlp_k6 --num-workers 4 --out runs/t2_seg'
# V-exact healing, k = 6 then k = 12.
bash -c 'W="cache/train_meta.json runs/m1/m1_structure.npz runs/x2/perlayer_reg_s0/results/e1_operators.npz"; for f in $W; do while [ ! -f $f ]; do sleep 60; done; done; export OMP_NUM_THREADS=4 OPENBLAS_NUM_THREADS=4 MKL_NUM_THREADS=4; python scripts/heal_vexact.py --config configs/m3_vexact.yaml --k 6 && python scripts/heal_vexact.py --config configs/m3_vexact.yaml --k 12 --out-dir runs/m3_vexact_5090'
# Generality: DINOv2-L (k = 12 of 24) and DINOv2-S (k = 6 of 12; the A100 run's files were not copied).
bash -c 'while [ ! -f cache/train_meta.json ]; do sleep 60; done; python scripts/run_e1.py --config configs/g_dinov2_l_reg.yaml --fit-seed 0 --ks 12 --out-root runs/g/dinov2_l_reg_s0'
bash -c 'while [ ! -f cache/train_meta.json ]; do sleep 60; done; python scripts/run_e1.py --config configs/g_dinov2_s_reg.yaml --fit-seed 0 --ks 6 --out-root runs/g/dinov2_s_reg_s0'
# Ablations: A1/A2 at k = 9 (seeds 0, 1, 2) and A4/A5 for seeds 1 and 2 (the A100 rows were not copied).
bash -c 'W="cache/train_meta.json runs/x2/perlayer_reg_s0/results/e1_operators.npz"; for f in $W; do while [ ! -f $f ]; do sleep 60; done; done; export OMP_NUM_THREADS=4 OPENBLAS_NUM_THREADS=4; python scripts/stage1_ablate.py --model dinov2_vitb14_reg --fit-run runs/x2/perlayer_reg_s0 --fit-seed 0 --ablation A1,A2 --ks 9 --out runs/a/dinov2_vitb14_reg'
bash -c 'W="cache/train_meta.json runs/x2/perlayer_reg_s1/results/e1_operators.npz"; for f in $W; do while [ ! -f $f ]; do sleep 60; done; done; export OMP_NUM_THREADS=4 OPENBLAS_NUM_THREADS=4; python scripts/stage1_ablate.py --model dinov2_vitb14_reg --fit-run runs/x2/perlayer_reg_s1 --fit-seed 1 --ablation A4,A5 --ks 6 --out runs/a/dinov2_vitb14_reg'
bash -c 'W="cache/train_meta.json runs/x2/perlayer_reg_s2/results/e1_operators.npz"; for f in $W; do while [ ! -f $f ]; do sleep 60; done; done; export OMP_NUM_THREADS=4 OPENBLAS_NUM_THREADS=4; python scripts/stage1_ablate.py --model dinov2_vitb14_reg --fit-run runs/x2/perlayer_reg_s2 --fit-seed 2 --ablation A4,A5 --ks 6 --out runs/a/dinov2_vitb14_reg'
bash -c 'W="cache/train_meta.json runs/x2/perlayer_reg_s1/results/e1_operators.npz"; for f in $W; do while [ ! -f $f ]; do sleep 60; done; done; export OMP_NUM_THREADS=4 OPENBLAS_NUM_THREADS=4; python scripts/stage1_ablate.py --model dinov2_vitb14_reg --fit-run runs/x2/perlayer_reg_s1 --fit-seed 1 --ablation A1,A2 --ks 9 --out runs/a/dinov2_vitb14_reg'
bash -c 'W="cache/train_meta.json runs/x2/perlayer_reg_s2/results/e1_operators.npz"; for f in $W; do while [ ! -f $f ]; do sleep 60; done; done; export OMP_NUM_THREADS=4 OPENBLAS_NUM_THREADS=4; python scripts/stage1_ablate.py --model dinov2_vitb14_reg --fit-run runs/x2/perlayer_reg_s2 --fit-seed 2 --ablation A1,A2 --ks 9 --out runs/a/dinov2_vitb14_reg'
# ADE20K, healed M3-pilot variants, in upload order.
bash -c 'for m in svd mirror random; do while [ ! -f runs/m3_pilot/$m/final/trained_blocks.pt ]; do sleep 60; done; python scripts/tier2_ade20k.py --model dinov2_vitb14_reg --variants healed_${m}_k9 --num-workers 4 --out runs/t2_seg || exit 1; done'
# Tier 2 for the healed V-exact models, once each run has finished.
bash -c 'while [ ! -f runs/m3_vexact/k6/SUMMARY.md ]; do sleep 120; done; python scripts/tier2_linear.py --model dinov2_vitb14_reg --variants healed_vexact_k6 --out runs/t2 && python scripts/tier2_transfer.py --model dinov2_vitb14_reg --variants healed_vexact_k6 --out runs/t2'
bash -c 'while [ ! -f runs/m3_vexact_5090/k12/SUMMARY.md ]; do sleep 120; done; python scripts/tier2_linear.py --model dinov2_vitb14_reg --variants healed_vexact_k12 --vexact-dir runs/m3_vexact_5090 --out runs/t2 && python scripts/tier2_transfer.py --model dinov2_vitb14_reg --variants healed_vexact_k12 --vexact-dir runs/m3_vexact_5090 --out runs/t2'
